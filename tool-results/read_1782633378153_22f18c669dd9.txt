   105→// Escape single quotes for SQL string literals.
   106→def esc($s) {
   107→    if (!$s) { return ""; }
   108→    string $out = "";
   109→    string $c = "";
   110→    number $i = 0;
   111→    while ($i < len($s)) {
   112→        $c = $s[$i];
   113→        if ($c == "'") {
   114→            $out = $out + "''";
   115→        } else {
   116→            $out = $out + $c;
   117→        }
   118→        $i = $i + 1;
   119→    }
   120→    return $out;
   121→}

   123→// Generate a pseudo-random hex token (32 chars = 128 bits).
   124→def newToken() {
   125→    string $t = "";
   126→    string $hex = "0123456789abcdef";
   127→    number $i = 0;
   128→    while ($i < 32) {
   129→        // random(15) returns 0..15 inclusive — perfect index into $hex
   130→        number $r = random(15);
   131→        $t = $t + $hex[$r];
   132→        $i = $i + 1;
   133→    }
   134→    return $t;
   135→}

   137→// Current epoch milliseconds (used as a monotonic message cursor).
   138→def nowMs() {
   139→    return clock();
   140→}

   142→// Look up the user by the Bearer token in $req.headers.authorization.
   143→// Returns a dict {id, username, displayName} or null.
   144→def authUser($req) {
   145→    dict $hdrs = $req.headers;
   146→    string $auth = "";
   147→    if ($hdrs.authorization) { $auth = $hdrs.authorization; }
   148→    if (len($auth) < 8) { return null; }
   149→    // Expect "Bearer <token>"
   150→    string $tok = substr($auth, 7);
   151→    if (len($tok) < 8) { return null; }
   152→
   153→    list $rows = sua.sqlite.query(
   154→        "SELECT id, username, display_name FROM users WHERE token = '" + esc($tok) + "';"
   155→    );
   156→    if (len($rows) == 0) { return null; }
   157→    dict $u = {
   158→        "id": num($rows[0].id),
   159→        "username": $rows[0].username,
   160→        "displayName": $rows[0].display_name
   161→    };
   162→    // Update last_seen
   163→    sua.sqlite.exec("UPDATE users SET last_seen = " + str(nowMs()) + " WHERE id = " + str($u.id) + ";");
   164→    return $u;
   165→}

   207→def handleRegister($req, $res) {
   208→    if (!$req.body.username || !$req.body.password) {
   209→        $res.status(400).json({"error": "username and password are required"});
   210→        return null;
   211→    }
   212→    string $username = $req.body.username;
   213→    string $password = $req.body.password;
   214→    string $displayName = $username;
   215→    if ($req.body.displayName) { $displayName = $req.body.displayName; }
   216→
   217→    list $exists = sua.sqlite.query("SELECT id FROM users WHERE username = '" + esc($username) + "';");
   218→    if (len($exists) > 0) {
   219→        $res.status(409).json({"error": "Username already taken"});
   220→        return null;
   221→    }
   222→
   223→    string $token = newToken();
   224→    dict $ins = sua.sqlite.exec(
   225→        "INSERT INTO users (username, password, display_name, token) VALUES ('" +
   226→        esc($username) + "', '" + esc($password) + "', '" + esc($displayName) + "', '" + esc($token) + "');"
   227→    );
   228→    number $uid = num($ins.lastInsertId);
   229→    $res.status(201).json({
   230→        "user": {"id": $uid, "username": $username, "displayName": $displayName},
   231→        "token": $token,
   232→        "message": "Account created"
   233→    });
   234→}

   280→def handleListUsers($req, $res) {

   289→        "(SELECT COUNT(*) FROM follows WHERE followee_id = u.id) AS followers_count, " +

   294→    number $i = 0;

   307→        $out[$i] = $u;

   313→def handleFollow($req, $res) {
   314→    dict $me = authUser($req);
   315→    if (!$me) {
   316→        $res.status(401).json({"error": "Not authenticated"});
   317→        return null;
   318→    }
   319→    number $target = num($req.params.id);
   320→    if ($target == $me.id) {
   321→        $res.status(400).json({"error": "Cannot follow yourself"});
   322→        return null;
   323→    }
   324→    list $exists = sua.sqlite.query(
   325→        "SELECT id FROM follows WHERE follower_id = " + str($me.id) + " AND followee_id = " + str($target) + ";"
   326→    );
   327→    if (len($exists) > 0) {
   328→        $res.json({"following": true, "message": "Already following"});
   329→        return null;
   330→    }
   331→    sua.sqlite.exec(
   332→        "INSERT INTO follows (follower_id, followee_id) VALUES (" + str($me.id) + ", " + str($target) + ");"
   333→    );
   334→    list $tu = sua.sqlite.query("SELECT username, display_name FROM users WHERE id = " + str($target) + ";");
   335→    if (len($tu) > 0) {
   336→        notify($target, "follow", $me.displayName + " started following you.", "/u/" + $me.username);
   337→    }
   338→    $res.json({"following": true, "message": "Followed"});
   339→}

   417→def handleLikePost($req, $res) {

   419→    if (!$me) {
   420→        $res.status(401).json({"error": "Not authenticated"});
   421→        return null;
   422→    }
   423→    number $pid = num($req.params.id);
   424→    list $exists = sua.sqlite.query(
   425→        "SELECT id FROM likes WHERE post_id = " + str($pid) + " AND user_id = " + str($me.id) + ";"
   426→    );
   427→    if (len($exists) > 0) {
   428→        sua.sqlite.exec("DELETE FROM likes WHERE post_id = " + str($pid) + " AND user_id = " + str($me.id) + ";");
   429→        list $cnt = sua.sqlite.query("SELECT COUNT(*) AS n FROM likes WHERE post_id = " + str($pid) + ";");
   430→        $res.json({"liked": false, "likes": num($cnt[0].n)});
   431→        return null;
   432→    }
   433→    sua.sqlite.exec(
   434→        "INSERT INTO likes (post_id, user_id) VALUES (" + str($pid) + ", " + str($me.id) + ");"
   435→    );
   436→    list $owner = sua.sqlite.query("SELECT user_id FROM posts WHERE id = " + str($pid) + ";");
   437→    if (len($owner) > 0 && num($owner[0].user_id) != $me.id) {
   438→        notify(num($owner[0].user_id), "like", $me.displayName + " liked your post.", "/post/" + str($pid));

   574→def handleConversations($req, $res) {
   575→    dict $me = authUser($req);
   576→    if (!$me) {
   577→        $res.status(401).json({"error": "Not authenticated"});
   578→        return null;
   579→    }
   580→    // Last message per peer — use UNION to gather both directions,
   581→    // then pick the max id per peer_id.
   582→    list $rows = sua.sqlite.query(
   583→        "SELECT m.id, m.from_id, m.to_id, m.body, m.created_at, m.delivered, " +
   584→        "  CASE WHEN m.from_id = " + str($me.id) + " THEN m.to_id ELSE m.from_id END AS peer_id, " +
   585→        "  u.username AS peer_username, u.display_name AS peer_display_name, u.last_seen AS peer_last_seen " +
   586→        "FROM messages m " +
   587→        "JOIN users u ON u.id = (CASE WHEN m.from_id = " + str($me.id) + " THEN m.to_id ELSE m.from_id END) " +
   588→        "WHERE m.id IN (" +
   589→        "  SELECT MAX(id) FROM (" +
   590→        "    SELECT id, from_id AS peer_id FROM messages WHERE to_id = " + str($me.id) +
   591→        "    UNION ALL" +
   592→        "    SELECT id, to_id AS peer_id FROM messages WHERE from_id = " + str($me.id) +
   593→        "  ) GROUP BY peer_id" +
   594→        ") ORDER BY m.id DESC;"
   595→    );
   596→    list $out = [];
   597→    number $i = 0;
   598→    each ($r in $rows) {
   599→        $out[$i] = {
   600→            "peer": {
   601→                "id": num($r.peer_id),
   602→                "username": $r.peer_username,
   603→                "displayName": $r.peer_display_name,
   604→                "online": (nowMs() - num($r.peer_last_seen)) < 60000
   605→            },
   606→            "lastMessage": {
   607→                "id": num($r.id),
   608→                "fromId": num($r.from_id),
   609→                "toId": num($r.to_id),
   610→                "body": $r.body,
   611→                "createdAt": $r.created_at,
   612→                "delivered": $r.delivered == 1
   613→            }
   614→        };
   615→        $i = $i + 1;
   616→    }
   617→    $res.json({"conversations": $out, "count": len($out)});
   618→}

   708→// ════════════════════════════════════════════════════════════════════
   709→//  WEBRTC SIGNALING — exchange offer/answer/ICE between two users.
   710→//  Flow:
   711→//    A: POST /api/call/offer/:to   {sdp}      →  store offer
   712→//    B: GET  /api/call/offer/:fromA          →  fetch offer (consumed=1)
   713→//    B: POST /api/call/answer/:toA  {sdp}     →  store answer
   714→//    A: GET  /api/call/answer/:fromB         →  fetch answer (consumed=1)
   715→//    A,B: POST /api/call/ice/:to  {candidate}  →  store ICE
   716→//    A,B: GET  /api/call/ice/:from            →  fetch unconsumed ICE
   717→//    A,B: POST /api/call/hangup/:to           →  store hangup signal
   718→// ════════════════════════════════════════════════════════════════════
   719→
   720→def handleCallOfferPost($req, $res) {
   721→    dict $me = authUser($req);
   722→    if (!$me) {
   723→        $res.status(401).json({"error": "Not authenticated"});
   724→        return null;
   725→    }
   726→    number $toId = num($req.params.id);
   727→    if (!$req.body.sdp) {
   728→        $res.status(400).json({"error": "sdp is required"});
   729→        return null;
   730→    }
   731→    // Remove any previous unconsumed offers from me to this peer
   732→    sua.sqlite.exec(
   733→        "DELETE FROM signaling WHERE from_id = " + str($me.id) + " AND to_id = " + str($toId) + " AND type = 'offer' AND consumed = 0;"
   734→    );
   735→    sua.sqlite.exec(
   736→        "INSERT INTO signaling (from_id, to_id, type, payload) VALUES (" + str($me.id) + ", " + str($toId) + ", 'offer', '" + esc($req.body.sdp) + "');"
   737→    );
   738→    notify($toId, "call", $me.displayName + " is calling you.", "/call/" + str($me.id));
   739→    $res.status(201).json({"ok": true, "message": "Offer sent"});
   740→}
