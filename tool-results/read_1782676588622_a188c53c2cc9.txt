    63→bool fileExists(const std::string& path) {
    64→    struct stat buffer;
    65→    return (stat(path.c_str(), &buffer) == 0);
    66→}

    68→// ─── Cross-platform filesystem helpers (C++17 <filesystem>) ───────────
    69→// These replace POSIX-only `system("mkdir -p ...")` / `system("rm -rf ...")`
    70→// / `system("cp ...")` calls so the installer works identically on Windows,
    71→// macOS, and Linux without needing a POSIX shell (no Git Bash / WSL required).
    72→
    73→namespace fs = std::filesystem;
    74→
    75→// Recursively create a directory (no-op if it already exists). Returns true
    76→// on success or if the directory already existed.
    77→bool fsMkdirP(const std::string& path) {
    78→    if (path.empty()) return true;
    79→    std::error_code ec;
    80→    fs::create_directories(fs::path(path), ec);
    81→    return !ec;
    82→}
    83→
    84→// Recursively remove a directory tree (no-op if it doesn't exist).
    85→bool fsRmRf(const std::string& path) {
    86→    if (path.empty()) return true;
    87→    std::error_code ec;
    88→    fs::remove_all(fs::path(path), ec);
    89→    return !ec;
    90→}
    91→
    92→// Copy a single file. Overwrites the destination if it exists.
    93→bool fsCopyFile(const std::string& src, const std::string& dst) {
    94→    std::error_code ec;
    95→    fs::copy_file(fs::path(src), fs::path(dst),
    96→                  fs::copy_options::overwrite_existing, ec);
    97→    return !ec;
    98→}
    99→
   100→// Recursively copy a directory tree. Creates dst if it doesn't exist.
   101→bool fsCopyDir(const std::string& src, const std::string& dst) {
   102→    std::error_code ec;
   103→    fs::copy(fs::path(src), fs::path(dst),
   104→             fs::copy_options::recursive | fs::copy_options::overwrite_existing,
   105→             ec);
   106→    return !ec;
   107→}
   108→
   109→// Copy all files matching a given extension from srcDir into dstDir.
   110→// Used to copy *.b source files into the Android assets folder without
   111→// relying on a shell glob.
   112→int fsCopyByExtension(const std::string& srcDir, const std::string& dstDir,
   113→                      const std::string& extWithDot) {
   114→    if (!fsMkdirP(dstDir)) return 0;
   115→    int copied = 0;
   116→    std::error_code ec;
   117→    for (auto& entry : fs::directory_iterator(fs::path(srcDir), ec)) {
   118→        if (ec) break;
   119→        if (!entry.is_regular_file()) continue;
   120→        if (entry.path().extension() == extWithDot) {
   121→            std::string dst = dstDir + "/" + entry.path().filename().string();
   122→            if (fsCopyFile(entry.path().string(), dst)) copied++;
   123→        }
   124→    }
   125→    return copied;
   126→}
   127→
   128→// Copy a single named file from srcDir to dstDir if it exists.
   129→bool fsCopyNamedIfExists(const std::string& srcDir, const std::string& dstDir,
   130→                         const std::string& filename) {
   131→    std::string src = srcDir + "/" + filename;
   132→    if (!fileExists(src)) return false;
   133→    if (!fsMkdirP(dstDir)) return false;
   134→    return fsCopyFile(src, dstDir + "/" + filename);
   135→}
   136→
   137→// Recursively copy a named subdirectory (e.g. "public") from srcDir into dstDir.
   138→bool fsCopyNamedDirIfExists(const std::string& srcDir, const std::string& dstDir,
   139→                            const std::string& dirname) {
   140→    std::string src = srcDir + "/" + dirname;
   141→    if (!fs::is_directory(fs::path(src))) return false;
   142→    if (!fsMkdirP(dstDir)) return false;
   143→    return fsCopyDir(src, dstDir + "/" + dirname);
   144→}

   146→std::string readFile(const std::string& path) {
   147→    std::ifstream file(path);
   148→    if (!file.is_open()) {
   149→        std::cerr << "  [ERROR] Cannot open file: " << path << "\n";
   150→        return "";
   151→    }
   152→    std::ostringstream ss;
   153→    ss << file.rdbuf();
   154→    return ss.str();
   155→}
   156→
   157→bool writeFile(const std::string& path, const std::string& content) {
   158→    std::ofstream file(path);
   159→    if (!file.is_open()) {
   160→        std::cerr << "  [ERROR] Cannot write to file: " << path << "\n";
   161→        return false;
   162→    }
   163→    file << content;
   164→    return true;
   165→}

   167→std::string getBasename(const std::string& path) {
   168→    size_t lastSlash = path.find_last_of("/\\");
   169→    std::string filename = (lastSlash == std::string::npos) ? path : path.substr(lastSlash + 1);
   170→    size_t lastDot = filename.find_last_of('.');
   171→    if (lastDot != std::string::npos) {
   172→        return filename.substr(0, lastDot);
   173→    }
   174→    return filename;
   175→}

   177→std::string getCurrentDir() {
   178→    char cwd[1024];
   179→    if (getcwd(cwd, sizeof(cwd)) != nullptr) {
   180→        return std::string(cwd);
   181→    }
   182→    return ".";
   183→}

   287→// ─── Run Code ─────────────────────────────────────────────────────────
   288→
   289→// v1.2.2: global quiet flag — flips evaluator into quiet mode and suppresses
   290→// the "Running:" / "[Executed in ...]" chit-chat. Set by --quiet on the CLI.
   291→static bool g_quietMode = false;
   292→
   293→void runCode(const std::string& source, const std::string& filename = "<repl>") {
   294→    auto start = std::chrono::high_resolution_clock::now();
   295→
   296→    Lexer lexer(source);
   297→    auto tokens = lexer.tokenize();
   298→
   299→    Parser parser(std::move(tokens));
   300→    auto ast = parser.parse();
   301→
   302→    Evaluator evaluator;
   303→    evaluator.setQuiet(g_quietMode);
   304→    // v1.2.1: tell the evaluator which file it's running so that
   305→    // `include "./..."` statements resolve relative to it.
   306→    if (filename != "<repl>") {
   307→        evaluator.setEntryPoint(filename);
   308→    }
   309→    evaluator.evaluate(ast);
   310→
   311→    auto end = std::chrono::high_resolution_clock::now();
   312→    auto us = std::chrono::duration_cast<std::chrono::microseconds>(end - start);
   313→    if (filename != "<repl>" && !g_quietMode) {
   314→        std::cout << "\n  [Executed in " << us.count() << " us]\n";
   315→    }
   316→}

   318→// ─── REPL ─────────────────────────────────────────────────────────────
   319→
   320→void runRepl() {
   321→    printBanner();
   322→
   323→    std::string line;
   324→    std::string buffer;
   325→
   326→    while (true) {
   327→        if (buffer.empty()) {
   328→            std::cout << "bantu> ";
   329→        } else {
   330→            std::cout << "  ...> ";
   331→        }
   332→
   333→        if (!std::getline(std::cin, line)) break;
   334→
   335→        size_t start = line.find_first_not_of(" \t\r\n");
   336→        if (start == std::string::npos) continue;
   337→        line = line.substr(start);
   338→
   339→        if (line == "exit" || line == "quit") {
   340→            std::cout << "  Goodbye! Kwaheri!\n";
   341→            break;
   342→        }
   343→        if (line == "help") {
   344→            printHelp();
   345→            continue;
   346→        }
   347→        if (line == "clear") {
   348→            buffer.clear();
   349→            continue;
   350→        }
   351→
   352→        buffer += line + "\n";
   353→
   354→        int braceCount = 0;
   355→        for (char c : buffer) {
   356→            if (c == '{') braceCount++;
   357→            if (c == '}') braceCount--;
   358→        }
   359→
   360→        if (braceCount <= 0) {
   361→            try {
   362→                runCode(buffer, "<repl>");
   363→            } catch (const std::exception& e) {
   364→                std::cerr << "  [ERROR] " << e.what() << "\n";
   365→            }
   366→            buffer.clear();
   367→        }
   368→    }
   369→}

   373→int cmdRun(const std::string& filePath) {
   374→    std::string path = filePath;
   375→
   376→    // If no file specified, look for main.b
   377→    if (path.empty()) {
   378→        if (fileExists("main.b")) {
   379→            path = "main.b";
   380→        } else if (fileExists("app.b")) {
   381→            path = "app.b";
   382→        } else if (fileExists("index.b")) {
   383→            path = "index.b";
   384→        } else {
   385→            std::cerr << "  [ERROR] No file specified and no main.b, app.b, or index.b found.\n";
   386→            std::cerr << "  Usage: bantu run <file.b> [--quiet]\n";
   387→            return 1;
   388→        }
   389→    }
   390→
   391→    if (!fileExists(path)) {
   392→        std::cerr << "  [ERROR] File not found: " << path << "\n";
   393→        return 1;
   394→    }
   395→
   396→    if (!g_quietMode) {
   397→        std::cout << "  Running: " << path << "\n";
   398→        std::cout << "  ────────────────────────────\n";
   399→    }
   400→
   401→    std::string source = readFile(path);
   402→    if (source.empty()) return 1;
   403→
   404→    try {
   405→        runCode(source, path);
   406→    } catch (const std::exception& e) {
   407→        std::cerr << "  [FATAL] " << e.what() << "\n";
   408→        return 1;
   409→    }
   410→
   411→    return 0;
   412→}

   416→std::string base64Encode(const std::string& input) {
   417→    static const char chars[] = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
   418→    std::string result;
   419→    int val = 0, valb = -6;
   420→    for (unsigned char c : input) {
   421→        val = (val << 8) + c;
   422→        valb += 8;
   423→        while (valb >= 0) {
   424→            result.push_back(chars[(val >> valb) & 0x3F]);
   425→            valb -= 6;
   426→        }
   427→    }
   428→    if (valb > -6) result.push_back(chars[((val << 8) >> (valb + 8)) & 0x3F]);
   429→    while (result.size() % 4) result.push_back('=');
   430→    return result;
   431→}

   433→int cmdBuild(const std::string& filePath) {
   434→    std::string path = filePath;
   435→
   436→    // If no file specified, look for main.b
   437→    if (path.empty()) {
   438→        if (fileExists("main.b")) {
   439→            path = "main.b";
   440→        } else if (fileExists("app.b")) {
   441→            path = "app.b";
   442→        } else if (fileExists("index.b")) {
   443→            path = "index.b";
   444→        } else {
   445→            std::cerr << "  [ERROR] No file specified and no main.b, app.b, or index.b found.\n";
   446→            std::cerr << "  Usage: bantu build <file.b>\n";
   447→            return 1;
   448→        }
   449→    }
   450→
   451→    if (!fileExists(path)) {
   452→        std::cerr << "  [ERROR] File not found: " << path << "\n";
   453→        return 1;
   454→    }
   455→
   456→    std::string basename = getBasename(path);
   457→    std::string outputPath = "./" + basename;
   458→
   459→    std::cout << "  Building: " << path << " -> " << outputPath << "\n";
   460→
   461→    // Read the source file
   462→    std::string source = readFile(path);
   463→    if (source.empty()) return 1;
   464→
   465→    // Encode source as base64 to avoid shell escaping issues
   466→    std::string encoded = base64Encode(source);
   467→
   468→    // Create a self-contained shell script
   469→    std::ostringstream script;
   470→    script << "#!/bin/bash\n";
   471→    script << "# Bantu Compiled Executable - " << basename << "\n";
   472→    script << "# Generated by bantu build v" << BANTU_VERSION << "\n";
   473→    script << "# Source: " << path << "\n";
   474→    script << "\n";
   475→
   476→    // Find the bantu interpreter
   477→    script << "BANTU_BIN=\"\"\n";
   478→    script << "if command -v bantu &> /dev/null; then\n";
   479→    script << "    BANTU_BIN=\"bantu\"\n";
   480→    script << "elif [ -f \"$HOME/.local/bin/bantu\" ]; then\n";
   481→    script << "    BANTU_BIN=\"$HOME/.local/bin/bantu\"\n";
   482→    script << "elif [ -f \"/usr/local/bin/bantu\" ]; then\n";
   483→    script << "    BANTU_BIN=\"/usr/local/bin/bantu\"\n";
   484→    script << "else\n";
   485→    script << "    echo \"[ERROR] Bantu interpreter not found. Install it first.\"\n";
   486→    script << "    echo \"  https://github.com/AsseySilivestir/swahiliscript\"\n";
   487→    script << "    exit 1\n";
   488→    script << "fi\n";
   489→    script << "\n";
   490→
   491→    // Embed base64-encoded source and decode at runtime
   492→    // Use temp file to avoid REPL mode (bantu with no args starts REPL)
   493→    script << "BANTU_TMPFILE=$(mktemp /tmp/bantu_XXXXXX.b)\n";
   494→    script << "echo '" << encoded << "' | base64 -d > \"$BANTU_TMPFILE\"\n";
   495→    script << "\"$BANTU_BIN\" run \"$BANTU_TMPFILE\"\n";
   496→    script << "rm -f \"$BANTU_TMPFILE\"\n";
   497→
   498→    if (writeFile(outputPath, script.str())) {

   906→// ─── bantu build-windows ──────────────────────────────────────────────
   907→// v1.2.1: Generate a Windows installer (.exe via NSIS) for the current
   908→// Bantu project. Requires makensis (NSIS) on PATH. The installer bundles:
   909→//   - bantu.exe (the interpreter)
   910→//   - the project's .b files
   911→//   - bantu.json manifest
   912→//   - a launcher .bat that runs `bantu run main.b`
   913→// On success, prints the path to the generated installer.
   914→//
   915→// Usage:
   916→//   bantu build-windows                 # uses main.b / app.b / index.b
   917→//   bantu build-windows <file.b>        # uses <file.b> as entry point
   918→//   bantu build-windows --name "MyApp"  # set installer app name
   919→//   bantu build-windows --version 1.0.0 # set installer version
   920→//
   921→// The generated installer is at ./dist/<Name>-Setup-<Version>.exe
   922→int cmdBuildWindows(int argc, char* argv[]) {
   923→    std::string entryFile;
   924→    std::string appName = "BantuApp";
   925→    std::string appVersion = "1.0.0";
   926→
   927→    for (int i = 2; i < argc; ++i) {
   928→        std::string a = argv[i];
   929→        if (a == "--name" && i + 1 < argc) {
   930→            appName = argv[++i];
   931→        } else if (a == "--version" && i + 1 < argc) {
   932→            appVersion = argv[++i];
   933→        } else if (a == "--help" || a == "-h") {
   934→            std::cout << "  Usage: bantu build-windows [options] [entry-file.b]\n";
   935→            std::cout << "  Options:\n";
   936→            std::cout << "    --name <Name>      Application name (default: BantuApp)\n";
   937→            std::cout << "    --version <x.y.z>  Application version (default: 1.0.0)\n";
   938→            std::cout << "\n  Requires NSIS (makensis) on PATH.\n";
   939→            std::cout << "  Output: ./dist/" << appName << "-Setup-" << appVersion << ".exe\n";
   940→            return 0;
   941→        } else if (!a.empty() && a[0] != '-') {
   942→            entryFile = a;
   943→        }
   944→    }
   945→
   946→    // Resolve entry file
   947→    if (entryFile.empty()) {
   948→        if (fileExists("main.b")) entryFile = "main.b";
   949→        else if (fileExists("app.b")) entryFile = "app.b";
   950→        else if (fileExists("index.b")) entryFile = "index.b";
   951→        else {
   952→            std::cerr << "  [ERROR] No entry file found (looked for main.b, app.b, index.b)\n";
   953→            return 1;
   954→        }
   955→    }
   956→    if (!fileExists(entryFile)) {
   957→        std::cerr << "  [ERROR] Entry file not found: " << entryFile << "\n";
   958→        return 1;
   959→    }
   960→
   961→    // Create dist/ directory
   962→    if (system("mkdir -p dist") != 0) {
   963→        std::cerr << "  [ERROR] Failed to create dist/ directory\n";
   964→        return 1;
   965→    }
   966→
   967→    // Write NSIS installer script
   968→    std::string nsiPath = "dist/installer.nsi";
   969→    std::ofstream nsi(nsiPath);
   970→    if (!nsi.is_open()) {
   971→        std::cerr << "  [ERROR] Cannot write " << nsiPath << "\n";
   972→        return 1;
   973→    }
   974→
   975→    nsi << "; Generated by bantu build-windows v" << BANTU_VERSION << "\n";
   976→    nsi << "!define APPNAME \"" << appName << "\"\n";
   977→    nsi << "!define APPVERSION \"" << appVersion << "\"\n";
   978→    nsi << "!define APPPUBLISHER \"Bantu\"\n";

  1048→// ─── bantu installer ─────────────────────────────────────────────────
  1049→// Cross-platform desktop installer generator.
  1050→//
  1051→//   bantu installer                        # auto-detect host platform
  1052→//   bantu installer app.b --platform linux
  1053→//   bantu installer app.b --platform windows
  1054→//   bantu installer app.b --platform macos
  1055→//
  1056→// Output goes to ./dist/. Each platform produces a runnable, offline-
  1057→// ready installer that bundles the bantu interpreter (so end users
  1058→// don't need Bantu installed) plus the project's .b source files and
  1059→// a launcher.
  1060→//
  1061→// Linux:   dist/<name>_<version>_amd64.deb  (dpkg-deb)
  1062→//          Also writes a .desktop file so the app appears in the
  1063→//          launcher menu.
  1064→//
  1065→// Windows: dist/<name>-Setup-<version>.exe  (NSIS .nsi)
  1066→//          If makensis is on PATH the .exe is built; otherwise the
  1067→//          .nsi is left for the user to compile.
  1068→//
  1069→// macOS:   dist/<Name>.app bundle
  1070→//          Contents/{MacOS,Resources} layout with Info.plist and a
  1071→//          launcher script. The .app can be opened with `open` or
  1072→//          double-clicked in Finder.
  1073→
  1074→// Trim leading/trailing whitespace
  1075→static std::string trimStr(const std::string& s) {
  1076→    size_t a = s.find_first_not_of(" \t\r\n");
  1077→    if (a == std::string::npos) return "";
  1078→    size_t b = s.find_last_not_of(" \t\r\n");
  1079→    return s.substr(a, b - a + 1);
  1080→}
  1081→
  1082→// Find a JSON string field by key. Very small JSON parser — enough
  1083→// for bantu.json which the scaffolder writes itself.
  1084→static std::string jsonField(const std::string& json, const std::string& key) {
  1085→    std::string needle = "\"" + key + "\"";
  1086→    size_t k = json.find(needle);
  1087→    if (k == std::string::npos) return "";
  1088→    size_t colon = json.find(':', k + needle.size());
  1089→    if (colon == std::string::npos) return "";
  1090→    size_t q1 = json.find('"', colon + 1);
  1091→    if (q1 == std::string::npos) return "";
  1092→    size_t q2 = json.find('"', q1 + 1);
  1093→    if (q2 == std::string::npos) return "";
  1094→    return json.substr(q1 + 1, q2 - q1 - 1);
  1095→}

  1097→// Locate the bantu interpreter to embed in the installer.
  1098→// Search order: $BANTU_BIN env, alongside current argv[0], PATH.
  1099→static std::string findSelfBinary() {
  1100→    const char* envBin = std::getenv("BANTU_BIN");
  1101→    if (envBin && *envBin && fileExists(envBin)) {
  1102→        return std::string(envBin);
  1103→    }
  1104→#ifdef _WIN32
  1105→    // Windows: GetModuleFileNameA returns the absolute path of the running .exe
  1106→    char buf[4096];
  1107→    DWORD n = GetModuleFileNameA(nullptr, buf, sizeof(buf) - 1);
  1108→    if (n > 0 && n < sizeof(buf)) {
  1109→        buf[n] = 0;
  1110→        return std::string(buf);
  1111→    }
  1112→    return "";
  1113→#elif defined(__APPLE__)
  1114→    // macOS: use _NSGetExecutablePath (from <mach-o/dyld.h>)
  1115→    // Returns the absolute path of the current process's executable,
  1116→    // even if invoked via a symlink or relative path.
  1117→    char buf[4096];
  1118→    uint32_t size = sizeof(buf);
  1119→    if (_NSGetExecutablePath(buf, &size) == 0) {
  1120→        // Resolve any symlinks in the path (e.g. /usr/local/bin/bantu → real location)
  1121→        char resolved[4096];
  1122→        if (realpath(buf, resolved) != nullptr) {
  1123→            return std::string(resolved);
  1124→        }
  1125→        return std::string(buf);
  1126→    }
  1127→    // Buffer too small — try with a heap allocation
  1128→    if (size > 0 && size < 1'000'000) {
  1129→        char* heapBuf = new char[size + 1];
  1130→        if (_NSGetExecutablePath(heapBuf, &size) == 0) {
  1131→            std::string result(heapBuf);
  1132→            delete[] heapBuf;
  1133→            char resolved[4096];
  1134→            if (realpath(result.c_str(), resolved) != nullptr) {
  1135→                return std::string(resolved);
  1136→            }
  1137→            return result;
  1138→        }
  1139→        delete[] heapBuf;
  1140→    }
  1141→    // Fall back to PATH lookup
  1142→    int rc = system("command -v bantu > /tmp/.bantu_bin_path 2>/dev/null");
  1143→    (void)rc;
  1144→    std::ifstream f("/tmp/.bantu_bin_path");
  1145→    if (f.is_open()) {
  1146→        std::string line;
  1147→        std::getline(f, line);
  1148→        if (!line.empty() && fileExists(line)) return line;
  1149→    }
  1150→    return "";
  1151→#else
  1152→    // Linux: /proc/self/exe is a symlink to the actual executable
  1153→    if (fileExists("/proc/self/exe")) {
  1154→        char buf[4096];
  1155→        ssize_t n = readlink("/proc/self/exe", buf, sizeof(buf) - 1);
  1156→        if (n > 0) { buf[n] = 0; return std::string(buf); }
  1157→    }
  1158→    // Fall back to PATH lookup
  1159→    int rc = system("command -v bantu > /tmp/.bantu_bin_path 2>/dev/null");
  1160→    (void)rc;
  1161→    std::ifstream f("/tmp/.bantu_bin_path");
  1162→    if (f.is_open()) {
  1163→        std::string line;
  1164→        std::getline(f, line);
  1165→        if (!line.empty() && fileExists(line)) return line;
  1166→    }
  1167→    return "";
  1168→#endif
  1169→}

  1171→// ── Linux: build a .deb package ──
  1172→static int buildLinuxInstaller(
  1173→    const std::string& entryFile,
  1174→    const std::string& appName,
  1175→    const std::string& appVersion,
  1176→    const std::string& iconPath,
  1177→    bool bundleBantu)
  1178→{
  1179→    std::string distDir = "dist";
  1180→    if (system(("mkdir -p " + distDir).c_str()) != 0) {
  1181→        std::cerr << "  [ERROR] Cannot create dist/\n";
  1182→        return 1;
  1183→    }
  1184→
  1185→    // Sanitize appName for package name (lowercase, no spaces).
  1186→    std::string pkgName = appName;
  1187→    for (auto& c : pkgName) {
  1188→        if (std::isalnum((unsigned char)c)) c = (char)std::tolower((unsigned char)c);
  1189→        else c = '_';
  1190→    }
  1191→    std::string pkgVer = appVersion;
  1192→    std::string stagingDir = "dist/" + pkgName + "_" + pkgVer + "_amd64_staging";
  1193→    std::string cleanCmd = "rm -rf '" + stagingDir + "'";
  1194→    system(cleanCmd.c_str());
  1195→
  1196→    // Standard .deb layout
  1197→    std::string debRoot = stagingDir + "/usr";
  1198→    std::string binDir       = debRoot + "/bin";
  1199→    std::string libDir       = debRoot + "/lib/" + pkgName;
  1200→    std::string shareDir     = debRoot + "/share/" + pkgName;
  1201→    std::string appsDir      = debRoot + "/share/applications";
  1202→    std::string iconsDir     = debRoot + "/share/icons/hicolor/256x256/apps";
  1203→
  1204→    system(("mkdir -p '" + binDir   + "'").c_str());
  1205→    system(("mkdir -p '" + libDir   + "'").c_str());
  1206→    system(("mkdir -p '" + shareDir + "'").c_str());
  1207→    system(("mkdir -p '" + appsDir  + "'").c_str());
  1208→    system(("mkdir -p '" + iconsDir + "'").c_str());
  1209→
  1210→    // ── lib/<pkg>/: copy all .b files (entry first), bantu.json, public/
  1211→    system(("cp *.b '" + libDir + "/' 2>/dev/null").c_str());
  1212→    system(("cp bantu.json '" + libDir + "/' 2>/dev/null").c_str());
  1213→    system(("cp -r public '" + libDir + "/' 2>/dev/null").c_str());
  1214→
  1215→    // ── Optional: bundle the bantu interpreter
  1216→    std::string bantuBinPath;
  1217→    if (bundleBantu) {
  1218→        bantuBinPath = findSelfBinary();
  1219→        if (!bantuBinPath.empty()) {
  1220→            system(("cp '" + bantuBinPath + "' '" + libDir + "/bantu' 2>/dev/null").c_str());
  1221→            system(("chmod +x '" + libDir + "/bantu'").c_str());
  1222→            std::cout << "  [INSTALLER] Bundled interpreter: " << bantuBinPath << "\n";
  1223→        } else {
  1224→            std::cerr << "  [WARN] Could not locate bantu binary to bundle.\n";
  1225→            std::cerr << "         Set $BANTU_BIN to point at it, or rerun without --bundle-bantu.\n";
  1226→        }
  1227→    }
  1228→
  1229→    // ── bin/<pkg>: launcher script (this is what users run from terminal)
  1230→    std::string launcherPath = binDir + "/" + pkgName;
  1231→    std::ofstream launcher(launcherPath);
  1232→    if (!launcher.is_open()) {
  1233→        std::cerr << "  [ERROR] Cannot write launcher: " << launcherPath << "\n";
  1234→        return 1;
  1235→    }
  1236→    launcher << "#!/bin/bash\n";
  1237→    launcher << "# Generated by bantu installer v" << BANTU_VERSION << "\n";
  1238→    launcher << "# App: " << appName << " v" << appVersion << "\n";
  1239→    launcher << "set -e\n";
  1240→    launcher << "APP_LIB=\"/usr/lib/" << pkgName << "\"\n";
  1241→    launcher << "if [ ! -d \"$APP_LIB\" ]; then\n";
  1242→    launcher << "    APP_LIB=\"$HOME/.local/lib/" << pkgName << "\"\n";
  1243→    launcher << "fi\n";
  1244→    launcher << "if [ -x \"$APP_LIB/bantu\" ]; then\n";
  1245→    launcher << "    BANTU_BIN=\"$APP_LIB/bantu\"\n";
  1246→    launcher << "elif command -v bantu >/dev/null 2>&1; then\n";
  1247→    launcher << "    BANTU_BIN=\"bantu\"\n";
  1248→    launcher << "else\n";
  1249→    launcher << "    echo \"[ERROR] Bantu interpreter not found. Install Bantu or use --bundle-bantu.\" >&2\n";
  1250→    launcher << "    exit 1\n";
  1251→    launcher << "fi\n";
  1252→    launcher << "cd \"$APP_LIB\"\n";
  1253→    launcher << "exec \"$BANTU_BIN\" run " << entryFile << " \"$@\"\n";
  1254→    launcher.close();
  1255→    system(("chmod +x '" + launcherPath + "'").c_str());
  1256→
  1257→    // ── share/applications/<pkg>.desktop — launcher menu entry
  1258→    std::string desktopPath = appsDir + "/" + pkgName + ".desktop";
  1259→    std::ofstream desktop(desktopPath);
  1260→    if (desktop.is_open()) {
  1261→        desktop << "[Desktop Entry]\n";
  1262→        desktop << "Version=" << appVersion << "\n";
  1263→        desktop << "Type=Application\n";
  1264→        desktop << "Name=" << appName << "\n";
  1265→        desktop << "Comment=" << appName << " (Bantu app)\n";
  1266→        desktop << "Exec=" << pkgName << "\n";
  1267→        if (!iconPath.empty() && fileExists(iconPath)) {
  1268→            // Copy the icon into the icons dir
  1269→            system(("cp '" + iconPath + "' '" + iconsDir + "/" + pkgName + ".png' 2>/dev/null").c_str());
  1270→            desktop << "Icon=" << pkgName << "\n";
  1271→        } else {
  1272→            desktop << "Icon=utilities-terminal\n";
  1273→        }
  1274→        desktop << "Terminal=true\n";
  1275→        desktop << "Categories=Development;Utility;\n";
  1276→        desktop.close();
  1277→    }
  1278→
  1279→    // ── DEBIAN/control
  1280→    std::string debianDir = stagingDir + "/DEBIAN";
  1281→    system(("mkdir -p '" + debianDir + "'").c_str());
  1282→    std::ofstream ctrl(debianDir + "/control");
  1283→    if (!ctrl.is_open()) {
  1284→        std::cerr << "  [ERROR] Cannot write DEBIAN/control\n";
  1285→        return 1;
  1286→    }
  1287→    ctrl << "Package: " << pkgName << "\n";
  1288→    ctrl << "Version: " << appVersion << "\n";
  1289→    ctrl << "Architecture: amd64\n";
  1290→    ctrl << "Maintainer: Assey Silivestir <assey@bantu-lang.dev>\n";
  1291→    ctrl << "Section: utils\n";
  1292→    ctrl << "Priority: optional\n";
  1293→    ctrl << "Description: " << appName << " — Bantu desktop application\n";
  1294→    ctrl << " Bundled Bantu interpreter + .b source files. Runs offline.\n";
  1295→    ctrl << " Built with bantu installer v" << BANTU_VERSION << ".\n";
  1296→    ctrl.close();
  1297→
  1298→    // ── DEBIAN/postinst — fix permissions after unpack
  1299→    std::ofstream postinst(debianDir + "/postinst");
  1300→    if (postinst.is_open()) {
  1301→        postinst << "#!/bin/bash\n";
  1302→        postinst << "set -e\n";
  1303→        postinst << "chmod +x /usr/bin/" << pkgName << "\n";
  1304→        postinst << "chmod -R +r /usr/lib/" << pkgName << " 2>/dev/null || true\n";
  1305→        postinst << "if [ -x /usr/lib/" << pkgName << "/bantu ]; then\n";
  1306→        postinst << "    chmod +x /usr/lib/" << pkgName << "/bantu\n";
  1307→        postinst << "fi\n";
  1308→        postinst << "exit 0\n";
  1309→        postinst.close();
  1310→        system(("chmod +x '" + debianDir + "/postinst'").c_str());
  1311→    }
  1312→
  1313→    // ── DEBIAN/prerm — pre-remove (no-op, just placeholder)
  1314→    std::ofstream prerm(debianDir + "/prerm");
  1315→    if (prerm.is_open()) {
  1316→        prerm << "#!/bin/bash\nset -e\nexit 0\n";
  1317→        prerm.close();
  1318→        system(("chmod +x '" + debianDir + "/prerm'").c_str());
  1319→    }
  1320→
  1321→    // ── Build the .deb
  1322→    std::string debName = pkgName + "_" + pkgVer + "_amd64.deb";
  1323→    std::string buildCmd = "dpkg-deb --build '" + stagingDir + "' 'dist/" + debName + "'";
  1324→    std::cout << "  [INSTALLER] Building .deb: " << debName << "\n";
  1325→    int rc = system(buildCmd.c_str());
  1326→    if (rc != 0) {
  1327→        std::cerr << "\n  [ERROR] dpkg-deb failed (rc=" << rc << ").\n";
  1328→        std::cerr << "  Staging dir left at: " << stagingDir << "\n";
  1329→        std::cerr << "  On non-Debian Linux, install 'dpkg-deb' or use the staging dir directly.\n";
  1330→        return 1;
  1331→    }
  1332→
  1333→    std::cout << "\n  [INSTALLER] Linux installer ready:\n";
  1334→    std::cout << "    dist/" << debName << "\n";
  1335→    std::cout << "\n  Install on any Debian/Ubuntu machine:\n";
  1336→    std::cout << "    sudo dpkg -i dist/" << debName << "\n";
  1337→    std::cout << "    " << pkgName << "              # run from terminal\n";
  1338→    std::cout << "    # Or find '" << appName << "' in your application launcher.\n";
  1339→    std::cout << "\n  Uninstall:\n";
  1340→    std::cout << "    sudo dpkg -r " << pkgName << "\n";
  1341→    return 0;
  1342→}

  1556→// End users get a signed-debug APK they can sideload on any Android 7.0+
  1557→// phone — the app runs fully offline with no Bantu installed on the device.
  1558→//
  1559→// If gradle is on PATH and ANDROID_HOME is set, we invoke `gradle assembleDebug`
  1560→// automatically; otherwise the user opens the project in Android Studio or
  1561→// runs `./build-apk.sh` once the SDK is installed.

  1563→// Sanitize appName into a valid Java package name (lowercase, dot-separated).
  1564→static std::string sanitizePkgName(const std::string& appName) {
  1565→    std::string s = appName;
  1566→    std::transform(s.begin(), s.end(), s.begin(),
  1567→                   [](unsigned char c){ return std::tolower(c); });
  1568→    std::string out;
  1569→    for (char c : s) {
  1570→        if ((c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') || c == '.') {
  1571→            out += c;
  1572→        } else if (c == ' ' || c == '-' || c == '_') {
  1573→            // word separator → dot
  1574→            if (!out.empty() && out.back() != '.') out += '.';
  1575→        }
  1576→    }
  1577→    // Strip leading/trailing dots
  1578→    while (!out.empty() && out.front() == '.') out.erase(out.begin());
  1579→    while (!out.empty() && out.back() == '.') out.pop_back();
  1580→    if (out.empty()) out = "bantu.app";
  1581→    // Must have at least one dot for Java package convention
  1582→    if (out.find('.') == std::string::npos) out = "dev.bantu." + out;
  1583→    // Don't start with a digit
  1584→    if (!out.empty() && out[0] >= '0' && out[0] <= '9') out = "dev." + out;
  1585→    return out;
  1586→}
  1587→
  1588→// Convert package name to a Java class name (PascalCase, stripped of dots).
  1589→static std::string sanitizeClassName(const std::string& appName) {
  1590→    std::string s = appName;
  1591→    std::string out;
  1592→    bool capitalizeNext = true;
  1593→    for (char c : s) {
  1594→        if ((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9')) {
  1595→            if (capitalizeNext && c >= 'a' && c <= 'z') {
  1596→                out += (char)(c - 32);
  1597→            } else {
  1598→                out += c;
  1599→            }
  1600→            capitalizeNext = false;
  1601→        } else {
  1602→            capitalizeNext = true;
  1603→        }
  1604→    }
  1605→    if (out.empty()) out = "BantuApp";
  1606→    // Don't start with a digit
  1607→    if (out[0] >= '0' && out[0] <= '9') out = "App" + out;
  1608→    return out;
  1609→}
